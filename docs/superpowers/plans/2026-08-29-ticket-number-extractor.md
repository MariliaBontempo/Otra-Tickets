**Goal:** Build a deterministic Python CLI that extracts one validated ticket number per PDF page and writes a JSON report containing the numbers, cross-page duplicates, and validation counts.

**Architecture:** Keep the utility in one focused Python module with pure functions for page parsing and report construction, plus thin functions for PDF reading, hashing, JSON writing, and command-line handling. `pypdf` supplies page text; the parser validates the two printed numeric values before report aggregation counts duplicates by page.

**Tech Stack:** Python 3.12, `pypdf` 6.11, standard library `argparse`, `hashlib`, `json`, `pathlib`, `collections`, and `unittest`.

#!/usr/bin/env python
import re


NUMERIC_PAIR_PATTERN = re.compile(r"^[ \t]*(\d+)[ \t]+(\d+)[ \t]*$")


def parse_page_number(text: str) -> str:
    pairs = [
        match.groups()
        for line in text.splitlines()
        if (match := NUMERIC_PAIR_PATTERN.fullmatch(line))
    ]
    return pairs[0][0]
```

class PageValidationError(ValueError):
    """Raised when a page does not contain one matching numeric pair."""


def parse_page_number(text: str) -> str:
    pairs = [
        match.groups()
        for line in text.splitlines()
        if (match := NUMERIC_PAIR_PATTERN.fullmatch(line))
    ]
    if not pairs:
        raise PageValidationError("missing numeric pair")
    if len(pairs) > 1:
        raise PageValidationError("multiple numeric pairs")

    left, right = pairs[0]
    if left != right:
        raise PageValidationError(
            f"mismatched printed values: {left!r} and {right!r}"
        )
    return left
```

```python
from collections import defaultdict
from collections.abc import Sequence


def build_report(
    *,
    source_filename: str,
    source_sha256: str,
    page_count: int,
    extracted: Sequence[tuple[int, str]],
    invalid_pages: Sequence[dict[str, object]],
) -> dict[str, object]:
    pages_by_number: dict[str, list[int]] = defaultdict(list)
    for page_number, number in extracted:
        pages_by_number[number].append(page_number)

    duplicates = [
        {
            "number": number,
            "occurrences": len(pages),
            "pages": pages,
        }
        for number, pages in pages_by_number.items()
        if len(pages) > 1
    ]
    invalid_page_list = list(invalid_pages)

    return {
        "source": {
            "filename": source_filename,
            "sha256": source_sha256,
        },
        "numbers": [number for _, number in extracted],
        "duplicates": duplicates,
        "validation": {
            "page_count": page_count,
            "valid_page_count": len(extracted),
            "invalid_page_count": len(invalid_page_list),
            "extracted_number_count": len(extracted),
            "unique_number_count": len(pages_by_number),
            "duplicate_number_count": len(duplicates),
            "duplicate_extra_occurrence_count": sum(
                duplicate["occurrences"] - 1 for duplicate in duplicates
            ),
            "invalid_pages": invalid_page_list,
        },
    }
```

import hashlib
import json
from pathlib import Path


def file_sha256(path: Path) -> str:
    digest = hashlib.sha256()
    with path.open("rb") as source:
        for chunk in iter(lambda: source.read(1024 * 1024), b""):
            digest.update(chunk)
    return digest.hexdigest()


def write_report(report: dict[str, object], output_path: Path) -> None:
    output_path.parent.mkdir(parents=True, exist_ok=True)
    output_path.write_text(
        json.dumps(report, indent=2, ensure_ascii=True) + "\n",
        encoding="utf-8",
    )
```

from collections.abc import Iterable


def extract_page_texts(
    page_texts: Iterable[str],
) -> tuple[list[tuple[int, str]], list[dict[str, object]]]:
    extracted: list[tuple[int, str]] = []
    invalid_pages: list[dict[str, object]] = []
    for page_number, text in enumerate(page_texts, start=1):
        try:
            extracted.append((page_number, parse_page_number(text)))
        except PageValidationError as error:
            invalid_pages.append({"page": page_number, "reason": str(error)})
    return extracted, invalid_pages
```

```python
def extract_pdf(
    input_path: Path,
) -> tuple[int, list[tuple[int, str]], list[dict[str, object]]]:
    reader = PdfReader(input_path)
    page_texts = ((page.extract_text() or "") for page in reader.pages)
    extracted, invalid_pages = extract_page_texts(page_texts)
    return len(reader.pages), extracted, invalid_pages


def parse_args(argv: Sequence[str] | None = None) -> argparse.Namespace:
    parser = argparse.ArgumentParser(
        description="Extract repeated ticket numbers from PDF pages."
    )
    parser.add_argument("input_pdf", type=Path)
    parser.add_argument("--output", required=True, type=Path)
    return parser.parse_args(argv)


def main(argv: Sequence[str] | None = None) -> int:
    args = parse_args(argv)
    try:
        page_count, extracted, invalid_pages = extract_pdf(args.input_pdf)
        report = build_report(
            source_filename=args.input_pdf.name,
            source_sha256=file_sha256(args.input_pdf),
            page_count=page_count,
            extracted=extracted,
            invalid_pages=invalid_pages,
        )
        write_report(report, args.output)
    except (OSError, PdfReadError) as error:
        print(f"could not process PDF: {error}", file=sys.stderr)
        return 2

    validation = report["validation"]
    print(
        f"wrote {args.output}: "
        f"{validation['extracted_number_count']} numbers, "
        f"{validation['duplicate_number_count']} duplicate values, "
        f"{validation['invalid_page_count']} invalid pages"
    )
    return 1 if invalid_pages else 0


if __name__ == "__main__":
    raise SystemExit(main())
```

# Ticket number extractor

Extract one validated numeric ticket value per PDF page from embedded text. The two
printed values on each page must match. Duplicate values are reported only when they
occur on different pages.

## Setup

Use the repository's configured Python and install the local dependency if needed:

```bash
python -m pip install -r scripts/ticket_number_extractor/requirements.txt

```bash
python scripts/ticket_number_extractor/extract_ticket_numbers.py input.pdf \
  --output output/ticket-numbers.json

The JSON contains `numbers`, `duplicates`, and `validation`. Numbers are strings to
preserve leading zeroes. Exit status 0 means every page validated, status 1 means the
report contains invalid pages, and status 2 means the input or output could not be
processed.
